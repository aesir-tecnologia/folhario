```typescript
// src/proxy.ts — Next 16 proxy file convention (L-1). Replaces deprecated middleware.ts.
// Source: https://nextjs.org/docs/app/api-reference/file-conventions/proxy
//         https://context7.com/amannn/next-intl (proxy.ts integration)
import createMiddleware from "next-intl/middleware";
import type { NextRequest } from "next/server";
import { routing } from "@i18n/routing";

const handleI18n = createMiddleware(routing);

export default function proxy(request: NextRequest) {
  const response = handleI18n(request);

  // D-24: minimum-viable security headers
  response.headers.set(
    "Strict-Transport-Security",
    "max-age=63072000; includeSubDomains; preload",
  );
  response.headers.set("X-Frame-Options", "DENY");
  response.headers.set("X-Content-Type-Options", "nosniff");

  return response;
}

export const config = {
  // Match everything EXCEPT API routes, _next internals, _vercel internals, and static assets.
  // API exclusion prevents next-intl from rewriting /api/* to locale-prefixed paths (Pitfall 9).
  matcher: "/((?!api|_next|_vercel|.*\\..*).*)",
};
```
