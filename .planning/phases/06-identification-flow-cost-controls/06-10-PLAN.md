<objective>
Wire the four Phase 6 route handlers — these are thin pass-throughs that auth + parse + idempotency-wrap + dispatch to use-cases. The non-trivial bit is the multipart POST handler which must (1) upload each photo to `identification-photos` bucket BEFORE calling `identify()`, (2) thread `requestStartMs` from the function entry, (3) declare `maxDuration = 60` for Vercel Pro (RESEARCH Pitfall 1).

Routes shipped:
- `POST /api/v1/identifications` — multipart, 1..5 photos, returns 201 with `{identification, results}`
- `GET /api/v1/identifications` — cursor-paginated, optional `?plantId=`
- `POST /api/v1/identifications/{id}/confirm` — JSON body, returns 200 `{plantId}`
- `POST /api/v1/identifications/{id}/correct` — JSON body, returns 200
- `POST /api/v1/iam/consents` — JSON body `{purpose: 'identification_third_party'}`, returns 201

Purpose: This plan closes the API surface so the frontend (Wave 5) has real endpoints to fetch + post against. Each route is unit-testable via Vitest integration project (it imports the route handler directly + mocks the multipart Request).

Output: Four route handlers + five integration test files.
</objective>

    **POST /api/v1/identifications/route.ts** — adapted from RESEARCH lines 718-801 with photo upload step BEFORE identify():
    ```typescript
    export const runtime = "nodejs";
    export const maxDuration = 60; // Vercel Pro required (RESEARCH Pitfall 1)

    import { randomUUID } from "node:crypto";

    export async function POST(request: Request): Promise<Response> {
      const auth = await requireVerifiedUser(request);
      if (!auth.ok) return errorResponse(auth.code, auth.code === ErrorCode.EmailUnverified ? "Verifique seu email para continuar." : "Sessão inválida.");

      const idempotencyKey = request.headers.get("idempotency-key")?.trim();
      if (!idempotencyKey) return errorResponse(ErrorCode.ValidationFailed, "Cabeçalho Idempotency-Key obrigatório.");

      const requestStartMs = Date.now();
      const hashResult = await computeMultipartRequestHash(request);
      if (!hashResult.ok) return errorResponse(hashResult.code, hashResult.reason);

      let formData: FormData;
      try { formData = await request.formData(); }
      catch (e) { return errorResponse(ErrorCode.ValidationFailed, `multipart parse failed: ${e instanceof Error ? e.message : "unknown"}`); }

      const photoFiles = formData.getAll("photos").filter((v): v is File => v instanceof File);
      if (photoFiles.length === 0) return errorResponse(ErrorCode.ValidationFailed, "Pelo menos uma foto é obrigatória.");
      if (photoFiles.length > 5) return errorResponse(ErrorCode.ValidationFailed, "Máximo de 5 fotos por identificação.");

      // Validate each photo + collect buffers
      const validatedPhotos: Array<{ buffer: Buffer; contentType: string }> = [];
      for (const file of photoFiles) {
        const buf = Buffer.from(await file.arrayBuffer());
        const sizeCheck = rejectOversizeBuffer(buf);
        if (!sizeCheck.ok) return errorResponse(sizeCheck.code, sizeCheck.reason);
        const gpsCheck = await rejectGpsMetadata(buf);
        if (!gpsCheck.ok) return errorResponse(gpsCheck.code, gpsCheck.reason);
        validatedPhotos.push({ buffer: buf, contentType: file.type });
      }

      // BEFORE the use-case: upload each photo to identification-photos bucket.
      // We need a stable identificationId for the bucket key — generate it here and pass it to the use-case via input.
      // But the use-case generates the identification UUID at INSERT time. Two options:
      //   (a) generate id here, pass to use-case, use-case INSERT with that id.
      //   (b) two-step upload: use-case generates id, then route uploads after-the-fact.
      // Option (a) is simpler — the route owns the id. Keyspace `{userId}/{identificationId}/{n}.jpg`.
      const identificationId = randomUUID();
      const photosWithStoredUrls: Array<{ buffer: Buffer; contentType: string; storedUrl: string }> = [];
      for (let i = 0; i < validatedPhotos.length; i++) {
        const p = validatedPhotos[i]!;
        const upload = await uploadIdentificationPhoto({
          userId: auth.user.id,
          identificationId,
          index: i,
          buffer: p.buffer,
          contentType: p.contentType,
        });
        photosWithStoredUrls.push({ ...p, storedUrl: upload.storedUrl });
      }

      let capturedPostCommit: (() => Promise<void>) | undefined;
      const idempotencyResult = await withIdempotency(
        { userId: auth.user.id, key: idempotencyKey, requestHash: hashResult.hash },
        async (_tx) => {
          // identify() opens its own withUnitOfWork — does NOT use the idempotency tx
          const usecase = await identify({
            userId: auth.user.id,
            photos: photosWithStoredUrls,
            requestStartMs,
            // Pass the pre-generated id so the row INSERT uses it (use-case must be wired to accept this — verify by reading 06-08's IdentifyInput; if it doesn't, extend with `id?: string` field)
          });
          capturedPostCommit = usecase.postCommit;
          if (!usecase.ok) {
            const status = usecase.code === ErrorCode.CapHit ? 429
              : usecase.code === ErrorCode.ConsentRequired ? 403
              : usecase.code === ErrorCode.Timeout ? 504
              : usecase.code === ErrorCode.ProviderUnavailable ? 503 : 500;
            return { status, body: { error: { code: usecase.code, message: usecase.reason, ...(usecase.resetTime && { reset_time: usecase.resetTime }) } } };
          }
          return { status: 201, body: usecase.body };
        }
      );
      if (!idempotencyResult.replayed && capturedPostCommit) {
        try { await capturedPostCommit(); } catch (err) { Sentry.captureException(err); }
      }
      return new Response(JSON.stringify(idempotencyResult.body), {
        status: idempotencyResult.status,
        headers: { "content-type": "application/json" },
      });
    }
    ```

    ```typescript
    import { recordConsentRequestSchema } from "@contexts/identification/domain/schemas"; // 06-04

    export async function POST(request: Request): Promise<Response> {
      const auth = await requireVerifiedUser(request); if (!auth.ok) return errorResponse(auth.code, "...");
      const idempotencyKey = request.headers.get("idempotency-key")?.trim();
      if (!idempotencyKey) return errorResponse(ErrorCode.ValidationFailed, "...");
      const parsed = await parseJsonBody(request, recordConsentRequestSchema);
      if (!parsed.ok) return errorResponse(parsed.code, parsed.reason);

      const requestHash = createHash("sha256").update(JSON.stringify(parsed.value)).digest("hex");
      const result = await withIdempotency(
        { userId: auth.user.id, key: idempotencyKey, requestHash },
        async (tx) => {
          const inner = await recordConsent({ userId: auth.user.id, input: { purpose: "identification_third_party", legalBasis: "consent", source: "identify_screen" } }, tx);
          if (!inner.ok) return { status: 400, body: { error: { code: inner.error, message: inner.reason ?? "consent record failed" } } };
          // PostHog identification_consent_granted (server-side per D-26)
          getPostHog()?.capture({ distinctId: auth.user.id, event: "identification_consent_granted", properties: { source: "identify_screen" } });
          await getPostHog()?.shutdown();
          return { status: 201, body: { consent_version: inner.row.policyVersion?.version ?? null } };
        }
      );
      return new Response(JSON.stringify(result.body), { status: result.status, headers: { "content-type": "application/json" } });
    }
    ```
