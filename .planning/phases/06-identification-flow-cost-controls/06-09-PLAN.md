<objective>
TDD three companion use-cases that round out the identification flow:

1. **`confirmIdentification`** (IDENT-06): User selects a result → creates a Plant linked to Species + sets `Identification.plant_id` FK. Single TX (atomic — if Plant insert fails, Identification is unchanged). Emits `identification.succeeded` Inngest event in postCommit (D-18 + Phase 7 trigger contract).

2. **`correctIdentification`** (IDENT-07): User submits a manual species name; helper resolves it against `species.name` or `species.scientific_name` (case-insensitive); if hit, updates the linked plant's `species_id`; if miss, leaves null + flags Sentry. No blocking error.

3. **`listIdentifications`** (IDENT-08): Cursor-paginated user-scoped query; optional `plantId` filter; first photo URL signed with 24h TTL via `signIdentificationPhotoUrl` (06-01 helper); joined plant.name when applicable.

Purpose: 06-08 ships the identify use-case; this plan ships the three companion use-cases that close the IDENT-06 / IDENT-07 / IDENT-08 loop. They run in parallel with 06-08 (no shared file overlap inside `application/`).

Output: Three use-case files + three integration test files; ~12 tests covering happy path, error path, and edge cases per use-case.

    confirmIdentification — outer-tx pattern:
    ```typescript
    export async function confirmIdentification(input: {
      userId: string;
      identificationId: string;
      speciesId: string;
      providerSpeciesId: string;
      name: string;
      nickname?: string;
      location?: string;
      acquisitionDate?: string;
    }): Promise<ConfirmResult> {
      // Validate input via confirmIdentificationRequestSchema (06-04) before TX
      const parsed = confirmIdentificationRequestSchema.safeParse({ speciesId: input.speciesId, providerSpeciesId: input.providerSpeciesId, name: input.name, nickname: input.nickname, location: input.location, acquisitionDate: input.acquisitionDate });
      if (!parsed.success) return { ok: false, code: ErrorCode.ValidationFailed, reason: parsed.error.issues[0]?.message ?? "invalid" };

      return withUnitOfWork(input.userId, async (tx) => {
        // (1) Verify identification ownership + status
        const identification = await identificationsRepo.findByIdForUser(tx, input.userId, input.identificationId);
        if (!identification) return { ok: false, code: ErrorCode.NotFound, reason: "identification not found" };
        if (identification.status !== "success") return { ok: false, code: ErrorCode.ValidationFailed, reason: "cannot confirm a failed identification" };

        // (2) Verify selected result is one of the persisted provider results and is locally matched
        const selectedResult = (identification.results as IdentificationResult[]).find((r) => r.providerSpeciesId === input.providerSpeciesId);
        if (!selectedResult || selectedResult.speciesId !== input.speciesId) return { ok: false, code: ErrorCode.ValidationFailed, reason: "selected result is not confirmable" };

        // (3) Pre-fetch photo bytes from identification-photos bucket (pinned: Option A — re-upload via createPlant's buffer path).
        // Phase 5 createPlant takes {photo: {buffer, contentType}} and uploads to plant-photos. Confirm-identification re-encodes the
        // identification photo into the plant-photos bucket so plant.coverPhotoUrl points to plant-photos (single-bucket lifecycle).
        // Cost: one fetch + one upload per confirm. Acceptable — confirms are infrequent (one per successful identification).
        // Alternative options REJECTED: (B) cross-bucket reference mixes lifecycles; (C) extending createPlant with fromBucket mode
        // bloats Phase 5's use-case for a single Phase 6 caller. Re-upload keeps the contracts independent.
        const firstStoredUrl = identification.photoUrls[0];
        if (!firstStoredUrl) return { ok: false, code: ErrorCode.ValidationFailed, reason: "identification has no stored photo" };
        const signed = await signIdentificationPhotoUrl({ storedUrl: firstStoredUrl, ttlSeconds: 60 }); // 60s — bytes fetched immediately below
        if (!signed.ok) return { ok: false, code: ErrorCode.ProviderUnavailable, reason: "failed to sign identification photo" };
        const photoResp = await fetch(signed.signedUrl);
        if (!photoResp.ok) return { ok: false, code: ErrorCode.ProviderUnavailable, reason: `failed to fetch identification photo (${photoResp.status})` };
        const photoBuffer = Buffer.from(await photoResp.arrayBuffer());
        const photoContentType = photoResp.headers.get("content-type") ?? "image/jpeg";

        // (4) Outer-tx createPlant — pass the buffer (Phase 5 D-02 source:'identification' branch already supports speciesId)
        const plantResult = await createPlant({
          userId: input.userId,
          source: "identification",
          speciesId: input.speciesId,
          name: input.name.trim(),
          nickname: input.nickname?.trim(),
          location: input.location?.trim(),
          acquisitionDate: input.acquisitionDate,
          photo: { buffer: photoBuffer, contentType: photoContentType },
        }, { tx });
        if (!plantResult.ok) return { ok: false, code: plantResult.code, reason: plantResult.reason };

        // (5) Link identification → plant
        await identificationsRepo.updatePlantId(tx, input.identificationId, plantResult.plant.id, input.userId);
        await identificationsRepo.updateSelectedResult(tx, input.identificationId, selectedResult, input.userId);

        // (5) Build postCommit
        const postCommit = async () => {
          try {
            await inngest.send({
              name: "identification.succeeded",
              data: {
                identificationId: input.identificationId,
                userId: input.userId,
                plantId: plantResult.plant.id,
                provider: identification.provider,
                model: identification.model,
                selectedSpeciesId: input.speciesId,
                latencyMs: identification.latencyMs,
                succeededAt: new Date().toISOString(),
              } satisfies IdentificationSucceededPayload,
            });
            // createPlant's own postCommit (PostHog plant_added with source:'identification') ALREADY fires inside createPlant per Phase 5 D-28 — do NOT duplicate
            if (plantResult.postCommit) await plantResult.postCommit();
          } catch (err) { Sentry.captureException(err, { tags: { surface: "confirm-identification" } }); }
        };

        return { ok: true, plantId: plantResult.plant.id, postCommit };
      });
    }
    ```

    ```typescript
    export async function listIdentifications(input: { userId: string; cursor: string | null; limit: number; plantId?: string }): Promise<ListResult> {
      const limit = Math.min(Math.max(input.limit, 1), 200);
      let cursorPayload: ListByUserCursor | null = null;
      if (input.cursor) {
        const decoded = decodeSortCursor(input.cursor); // shared/api/cursor.ts from Phase 2
        if (!decoded.ok) return { ok: false, code: ErrorCode.ValidationFailed, reason: "invalid cursor" };
        cursorPayload = decoded.value;
      }
      const rows = await identificationsRepo.listByUser(defaultDb, input.userId, { cursor: cursorPayload, limit, plantId: input.plantId });
      // Join plant names for items where plantId is set
      const plantIds = rows.map(r => r.plantId).filter((id): id is string => !!id);
      const plantsMap = plantIds.length === 0 ? new Map() : new Map(
        (await defaultDb.select({ id: plants.id, name: plants.name }).from(plants).where(inArray(plants.id, plantIds))).map(p => [p.id, p.name])
      );
      // Sign first photo URL per row
      const signedItems = await Promise.all(rows.map(async (row) => {
        const firstPhoto = row.photoUrls?.[0];
        const signed = firstPhoto
          ? await signIdentificationPhotoUrl({ storedUrl: firstPhoto, ttlSeconds: 24 * 3600 })
          : null;
        return toIdentificationHistoryItemSnakeCase(row, signed?.ok ? signed.signedUrl : null, plantsMap.get(row.plantId ?? "") ?? null);
      }));
      const nextCursor = signedItems.length === limit
        ? encodeSortCursor({ createdAt: rows[rows.length - 1]!.createdAt, id: rows[rows.length - 1]!.id })
        : null;
      return { ok: true, items: signedItems, nextCursor };
    }
    ```
