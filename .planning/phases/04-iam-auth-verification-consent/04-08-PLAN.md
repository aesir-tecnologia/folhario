    **`src/contexts/iam/infrastructure/db/reset-tokens.ts`** — mirror verification-tokens.ts:
    ```typescript
    // Phase 4 D-09 + AUTH-11. Token hash storage; raw exists only in URL.
    import { eq, and, isNull, sql } from "drizzle-orm";
    import { db } from "@contexts/iam/infrastructure/db/client";
    import { passwordResetTokens } from "@contexts/iam/infrastructure/db/schema";
    import { mintToken } from "@shared/crypto/tokens";
    import { createHash } from "node:crypto";

    export async function revokeUnusedResetTokens(userId: string): Promise<void> {
      await db.update(passwordResetTokens)
        .set({ consumed_at: new Date() })
        .where(and(eq(passwordResetTokens.user_id, userId), isNull(passwordResetTokens.consumed_at)));
    }

    export async function mintResetToken(opts: {
      userId: string;
      sentToEmail: string;
    }): Promise<{ tokenId: string; rawToken: string }> {
      await revokeUnusedResetTokens(opts.userId);
      const { raw, hash } = mintToken();
      const expiresAt = new Date(Date.now() + 60 * 60 * 1000);  // D-09: 1h per AUTH-11
      const [row] = await db.insert(passwordResetTokens).values({
        user_id: opts.userId,
        token_hash: hash,
        expires_at: expiresAt,
        sent_to_email: opts.sentToEmail,
      }).returning({ id: passwordResetTokens.id });
      return { tokenId: row!.id, rawToken: raw };
    }

    export async function consumeResetToken(rawToken: string): Promise<{ userId: string } | null> {
      const hash = createHash("sha256").update(rawToken).digest("hex");
      const result = await db.execute<{ user_id: string }>(
        sql`WITH t AS (
              SELECT id, user_id FROM public.password_reset_tokens
              WHERE token_hash = ${hash} AND consumed_at IS NULL AND expires_at > now()
              FOR UPDATE
            )
            UPDATE public.password_reset_tokens prt SET consumed_at = now()
            FROM t
            WHERE prt.id = t.id
            RETURNING prt.user_id`
      );
      const row = result[0];
      return row ? { userId: row.user_id } : null;
    }
    ```

    ```typescript
    // Phase 4 D-10 + AUTH-12. Validate + consume token + admin.updateUserById.
    // CRITICAL: do NOT call signOut or revokeRefreshToken — existing JWTs MUST remain valid.
    import { consumeResetToken } from "@contexts/iam/infrastructure/db/reset-tokens";
    import { supabaseAdmin } from "@contexts/iam/infrastructure/supabase-admin";

    export type ConsumeResult =
      | { kind: "success" }
      | { kind: "invalid_or_expired" };

    export async function consumePasswordReset(opts: {
      token: string;
      password: string;
    }): Promise<ConsumeResult> {
      const consumed = await consumeResetToken(opts.token);
      if (!consumed) return { kind: "invalid_or_expired" };

      const { error } = await supabaseAdmin.auth.admin.updateUserById(consumed.userId, {
        password: opts.password,
      });
      if (error) {
        // Token is already consumed at this point — surface as internal error.
        // The user can request a new reset. Token reuse is impossible (consumed_at is set).
        throw new Error(`updateUserById failed: ${error.message}`);
      }
      return { kind: "success" };
    }
    ```

    ```typescript
    import { NextResponse } from "next/server";
    import { errorResponse, ErrorCode } from "@shared/config/errors";
    import { withThrottle } from "@shared/api/throttle";
    import { passwordResetRequestSchema } from "@contexts/iam/domain/schemas";
    import { requestPasswordReset } from "@contexts/iam/application/request-password-reset";

    export async function POST(request: Request) {
      return withThrottle(request, "password-reset-request", "always", async () => {
        let body: unknown;
        try {
          body = await request.json();
        } catch {
          // Even malformed body returns 200 (anti-enumeration).
          return NextResponse.json({ ok: true, message: "Verifique seu e-mail." });
        }

        const parsed = passwordResetRequestSchema.safeParse(body);
        if (!parsed.success) {
          // Same 200 response — no signal whether email is valid or not.
          return NextResponse.json({ ok: true, message: "Verifique seu e-mail." });
        }

        await requestPasswordReset(parsed.data.email, new URL(request.url));
        return NextResponse.json({ ok: true, message: "Verifique seu e-mail." });
      });
    }
    ```
