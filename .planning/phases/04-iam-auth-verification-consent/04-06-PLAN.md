<objective>
Implement the email+password signup flow per resolved Q1 + CONTEXT D-03 + D-25 + D-32 (PartnerStore validation), the verification gate enforcement on gated endpoints, the verification link consumption flow at /auth/verify (per Codex HIGH #2: set email_verified_at FIRST then consume token in one tx), and the resend-verification endpoint. This plan covers the heaviest IAM use-case and ships the real `users` repository (replacing Plan 03's stub), four new repositories (consent-logs, policy-versions, subscriptions, verification-tokens) plus the new `partner-store.ts` repo (Codex MEDIUM partner_code fix per D-32), and the PostHog server-side bridge.

**Codex HIGH #2 fix — Transactional signup:** Plan 06 wraps the entire DB-side signup work in ONE `db.transaction(async (tx) => { ... })` block. All repos that participate in signup accept an optional `dbOrTx` parameter (defaults to the singleton `db`); the signup use-case passes the transaction handle so all writes commit atomically. If any DB write fails, the tx rolls back AND a compensating `authAdapter.adminDeleteUser(userId)` runs. The verify-email use-case ALSO wraps `setEmailVerifiedAt` + `consumeVerificationToken` in a single tx, with the order being SET email_verified_at FIRST then CONSUME token (so a failure between the two leaves the user verified — preferable to leaving them unverified with a consumed token).

**Codex HIGH #3 fix — Adapter Boundary:** This plan does NOT call `supabase.auth.*` directly. All auth interactions go through `authAdapter` from Plan 03 (`authAdapter.createUser`, `authAdapter.signInWithPassword`, `authAdapter.adminDeleteUser`). The route handler is a thin shell.

**Codex HIGH #6 fix — E2E with real cookies:** The signup-verification flow is exercised end-to-end via Playwright (`tests/e2e/auth-signup-verification.spec.ts`) against a running Next dev server. The Playwright test asserts: signup → Resend mock captures verification email → click verify link → cookie set → GET /api/v1/iam/me returns 200 (proving the gate sees the user as verified). Direct route-function tests for cookie/session-bearing flows are removed in favor of this E2E. Pure repository tests (without cookies/sessions) stay in vitest.

**D-32 fix — PartnerStore validation:** signupUser looks up `input.partner_code` against `partner_store` table per Plan 02's schema. Active row → trial_source='partner' + 30 days. Inactive or absent row → reject with `invalid_partner_code` 422 (already in the closed error registry). Empty/null partner_code → trial_source='organic' + 14 days. Plan 09 oauth-complete uses the same path.

**D-31 fix — JSON-only routes:** All Phase 4 auth route handlers consume `request.json()` (NOT `request.formData()`). UI in Plan 10 uses client components with `fetch({headers: {'Content-Type': 'application/json'}})`.

Purpose: AUTH-01 (signup), AUTH-02 (verification gate), AUTH-04 (verify link sets email_verified_at), AUTH-06 (age ≥13), AUTH-07 (partner code captured + validated per D-32), AUTH-08 (timezone captured), AUTH-09 (ConsentLog × 2 with active policy version) — the core "create a verified Brazilian beginner account in pt-BR" loop.

Output: A working signup → verify → app-shell flow, with all required repositories (including partner-store.ts), 4 vitest integration tests + 1 Playwright E2E covering the full cookie-bearing flow, and atomic transaction semantics with compensating delete on failure.

    import { eq, sql } from "drizzle-orm";
    import { db } from "@contexts/iam/infrastructure/db/client";
    import { users } from "@contexts/iam/infrastructure/db/schema";
    import type { User } from "@contexts/iam/domain/schemas";
    import type { DbOrTx } from "@contexts/iam/infrastructure/db/types";

    export async function getUserById(id: string): Promise<User | null> {
      const rows = await db.execute<{ id: string; email: string; email_verified_at: Date | null; age_confirmed_at: Date | null; timezone: string; partner_code: string | null; has_password: boolean }>(
        sql`SELECT pu.id, pu.email, pu.email_verified_at, pu.age_confirmed_at, pu.timezone, pu.partner_code,
                   (au.encrypted_password IS NOT NULL) AS has_password
            FROM public.users pu
            JOIN auth.users au ON au.id = pu.id
            WHERE pu.id = ${id}`
      );
      const row = rows[0];
      if (!row) return null;
      return {
        id: row.id,
        email: row.email,
        emailVerifiedAt: row.email_verified_at,
        ageConfirmedAt: row.age_confirmed_at,
        timezone: row.timezone,
        partnerCode: row.partner_code,
        hasPassword: row.has_password,
      };
    }

    export async function getUserByEmail(email: string): Promise<User | null> {
      const rows = await db.execute<{ id: string; email: string; email_verified_at: Date | null; age_confirmed_at: Date | null; timezone: string; partner_code: string | null; has_password: boolean }>(
        sql`SELECT pu.id, pu.email, pu.email_verified_at, pu.age_confirmed_at, pu.timezone, pu.partner_code,
                   (au.encrypted_password IS NOT NULL) AS has_password
            FROM public.users pu
            JOIN auth.users au ON au.id = pu.id
            WHERE pu.email = ${email}`
      );
      const row = rows[0];
      if (!row) return null;
      return {
        id: row.id,
        email: row.email,
        emailVerifiedAt: row.email_verified_at,
        ageConfirmedAt: row.age_confirmed_at,
        timezone: row.timezone,
        partnerCode: row.partner_code,
        hasPassword: row.has_password,
      };
    }

    export async function insertPublicUser(input: {
      id: string; email: string; age_confirmed_at: Date; timezone: string; partner_code: string | null;
    }, dbOrTx: DbOrTx = db): Promise<void> {
      await dbOrTx.execute(
        sql`INSERT INTO public.users (id, email, age_confirmed_at, timezone, partner_code)
            VALUES (${input.id}, ${input.email}, ${input.age_confirmed_at}, ${input.timezone}, ${input.partner_code})
            ON CONFLICT (id) DO UPDATE SET
              email = EXCLUDED.email,
              age_confirmed_at = EXCLUDED.age_confirmed_at,
              timezone = EXCLUDED.timezone,
              partner_code = EXCLUDED.partner_code`
      );
    }

    export async function setEmailVerifiedAt(id: string, when: Date, dbOrTx: DbOrTx = db): Promise<void> {
      await dbOrTx.update(users).set({ email_verified_at: when }).where(eq(users.id, id));
    }

    import { db } from "@contexts/iam/infrastructure/db/client";
    import { emailVerificationTokens } from "@contexts/iam/infrastructure/db/schema";
    import { mintToken } from "@shared/crypto/tokens";
    import { createHash } from "node:crypto";
    import type { DbOrTx } from "@contexts/iam/infrastructure/db/types";

    export async function revokeUnusedVerificationTokens(userId: string, dbOrTx: DbOrTx = db): Promise<void> {
      await dbOrTx.update(emailVerificationTokens)
        .set({ consumed_at: new Date() })
        .where(and(eq(emailVerificationTokens.user_id, userId), isNull(emailVerificationTokens.consumed_at)));

    export async function mintVerificationToken(opts: {
      userId: string;
      sentToEmail: string;
    }, dbOrTx: DbOrTx = db): Promise<{ tokenId: string; rawToken: string }> {
      await revokeUnusedVerificationTokens(opts.userId, dbOrTx);
      const { raw, hash } = mintToken();
      const expiresAt = new Date(Date.now() + 24 * 60 * 60 * 1000);
      const [row] = await dbOrTx.insert(emailVerificationTokens).values({
        user_id: opts.userId,
        token_hash: hash,
        expires_at: expiresAt,
        sent_to_email: opts.sentToEmail,
      }).returning({ id: emailVerificationTokens.id });
      return { tokenId: row!.id, rawToken: raw };
    }

    export async function consumeVerificationToken(rawToken: string, dbOrTx: DbOrTx = db): Promise<{ userId: string } | null> {
      const hash = createHash("sha256").update(rawToken).digest("hex");
      const result = await dbOrTx.execute<{ user_id: string }>(
        sql`WITH t AS (
              SELECT id, user_id FROM public.email_verification_tokens
              WHERE token_hash = ${hash} AND consumed_at IS NULL AND expires_at > now()
              FOR UPDATE
            )
            UPDATE public.email_verification_tokens evt SET consumed_at = now()
            FROM t
            WHERE evt.id = t.id
            RETURNING evt.user_id`
      );
      const row = result[0];
      return row ? { userId: row.user_id } : null;
    }

    import { bumpThrottle, isLocked } from "@contexts/iam/infrastructure/db/auth-throttle";

        return NextResponse.json({ ok: true });
