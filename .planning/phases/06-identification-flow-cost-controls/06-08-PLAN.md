    ```typescript
    // src/contexts/identification/application/identify.ts
    import * as Sentry from "@sentry/nextjs";
    import { sql } from "drizzle-orm";
    import { ErrorCode } from "@shared/config/errors";
    import { withUnitOfWork, type TransactionalDb } from "@shared/db/unit-of-work";
    import { defaultDb } from "@shared/db/client";
    import { inngest } from "@shared/inngest/client";
    import { getPostHog } from "@shared/telemetry/posthog-server";

    import { IDENT_LOCK_NAMESPACE } from "@contexts/identification/domain/locks";
    import { type IdentificationFailureReason, type IdentificationFailedPayload } from "@contexts/identification/domain/events";
    import { dispatchIdentification, type RouterDeps } from "./identification-router";
    import { getIdentificationProviders } from "@contexts/identification/infrastructure/providers";

    import * as identificationsRepo from "@contexts/identification/infrastructure/db/identifications";
    import * as limitsRepo from "@contexts/identification/infrastructure/db/identification-limits";
    import * as budgetsRepo from "@contexts/identification/infrastructure/db/provider-budgets";
    import * as countersRepo from "@contexts/identification/infrastructure/db/provider-usage-counters";
    import * as breakerRepo from "@contexts/identification/infrastructure/db/circuit-breakers";
    import * as consentLogsRepo from "@contexts/iam/infrastructure/db/consent-logs";
    import * as policyVersionsRepo from "@contexts/iam/infrastructure/db/policy-versions";
    import * as subscriptionsRepo from "@contexts/iam/infrastructure/db/subscriptions";
    import { toIdentificationSnakeCase } from "@contexts/identification/api/snake-case";

    const POLICY_VERSION_TTL_MS = 30_000;
    let cachedPolicyVersion: { version: string; expiresAt: number } | null = null;
    async function getCachedPolicyVersionString(db) {
      const now = Date.now();
      if (cachedPolicyVersion && cachedPolicyVersion.expiresAt > now) return cachedPolicyVersion.version;
      const v = await policyVersionsRepo.getCurrentPolicyVersionString(db, "privacy_policy");
      cachedPolicyVersion = { version: v, expiresAt: now + POLICY_VERSION_TTL_MS };
      return v;
    }
    /** Test-only — clears cache between tests. */
    export function __resetPolicyVersionCacheForTests() { cachedPolicyVersion = null; }

    function nextUtcMidnightIso(): string {
      const tomorrow = new Date();
      tomorrow.setUTCDate(tomorrow.getUTCDate() + 1);
      tomorrow.setUTCHours(0, 0, 0, 0);
      return tomorrow.toISOString();
    }

    export interface IdentifyInput {
      userId: string;
      photos: ReadonlyArray<{ buffer: Buffer; contentType: string; storedUrl: string }>;
      requestStartMs: number;
      /**
       * Pre-generated identification id. Route handler (06-10) generates the UUID before
       * uploading photos to identification-photos so the bucket key path matches the row id.
       * If undefined (e.g. unit tests), the use-case lets DB default generate a new UUID.
       */
      identificationId?: string;
    }
    export type IdentifyResult =
      | { ok: true; body: { identification: ReturnType<typeof toIdentificationSnakeCase>; results: ResultSnakeCase[] }; postCommit: () => Promise<void> }
      | { ok: false; code: ErrorCode; reason: string; resetTime?: string; postCommit?: () => Promise<void> };

    export async function identify(input: IdentifyInput): Promise<IdentifyResult> {
      const { userId, photos, requestStartMs } = input;
      const utcDate = new Date().toISOString().slice(0, 10);

      return withUnitOfWork(userId, async (tx) => {
        // Step 1: Advisory lock per user (D-05)
        const lockResult = await tx.execute<{ ok: boolean }>(sql`
          SELECT pg_try_advisory_xact_lock(${IDENT_LOCK_NAMESPACE}, hashtext(${userId})) AS ok
        `);
        if (!lockResult[0]?.ok) {
          return { ok: false, code: ErrorCode.CapHit, reason: "concurrent identification rejected" };
        }

        // Step 2: Consent gate (D-09 / LGPD-09)
        const consented = await consentLogsRepo.hasActiveConsent(tx, userId, "identification_third_party");
        if (!consented) {
          return { ok: false, code: ErrorCode.ConsentRequired, reason: "identification_third_party consent missing" };
        }

        // Step 3: Cap check (D-04 / COST-01 / COST-02 / COST-03)
        const sub = await subscriptionsRepo.findActiveByUserId(tx, userId);
        const tier = sub?.status === "active" ? "paid" : "trial";
        const limit = await limitsRepo.findByTier(tx, tier);
        const dailyCount = await identificationsRepo.countDailyForUser(tx, userId, utcDate);
        if (dailyCount >= limit.dailyCap) {
          return { ok: false, code: ErrorCode.CapHit, reason: "daily cap reached", resetTime: nextUtcMidnightIso() };
        }
        const windowStart = tier === "trial" ? sub?.trialStartDate : sub?.currentPeriodStart;
        const windowEnd = tier === "trial" ? sub?.trialEndDate : sub?.currentPeriodEnd;
        if (windowStart && windowEnd) {
          const periodCount = await identificationsRepo.countInWindowForUser(tx, userId, windowStart, windowEnd);
          if (periodCount >= limit.periodCap) {
            return { ok: false, code: ErrorCode.CapHit, reason: "period cap reached", resetTime: windowEnd };
          }
        }

        // Step 4: Resolve current policy_version STRING (D-11)
        const consentVersion = await getCachedPolicyVersionString(tx);

        // Step 5: Router dispatch (06-06)
        const routerDeps: RouterDeps = {
          providers: getIdentificationProviders(),
          breakerRepo: {
            findByProvider: (p) => breakerRepo.findByProvider(defaultDb, p),
            recordSuccess: async (p) => { await breakerRepo.recordSuccess(defaultDb, p); },
            recordFailure: async (p) => { await breakerRepo.recordFailure(defaultDb, p); },
            transitionToHalfOpen: async (p) => { await breakerRepo.transitionToHalfOpen(defaultDb, p); },
          },
          budgetRepo: {
            findByProviderAndPurpose: (p, purpose) => budgetsRepo.findByProviderAndPurpose(defaultDb, p, purpose),
            findUsageCounter: async (p, purpose, date) => {
              const r = await countersRepo.findByProviderAndPurposeToday(defaultDb, p, purpose, date);
              return r ? { estimatedCostCents: r.estimatedCostCents } : null;
            },
          },
        };
        const routerInput = {
          userId,
          photos: photos.map(p => ({ buffer: p.buffer, contentType: p.contentType })),
          requestStartMs,
        };
        const routerResult = await dispatchIdentification(routerDeps, routerInput);

        // Step 6: Persist Identification row (success OR failure path — IDENT-12 / IDENT-14)
        const photoUrls = photos.map(p => p.storedUrl);
        const insertedRow = routerResult.ok
          ? await identificationsRepo.create(tx, {
              ...(input.identificationId ? { id: input.identificationId } : {}),
              userId, photoUrls,
              provider: routerResult.result.provider,
              model: routerResult.result.model,
              results: routerResult.result.results,
              latencyMs: routerResult.result.latencyMs,
              consentVersion, status: "success", failureReason: null,
            })
          : await identificationsRepo.create(tx, {
              ...(input.identificationId ? { id: input.identificationId } : {}),
              userId, photoUrls,
              provider: routerResult.provider || "unknown",
              model: "",
              results: [],
              latencyMs: 0,
              consentVersion, status: "failed",
              failureReason: routerResult.failureReason,
            });

        // Step 7: Build post-commit telemetry hooks (run AFTER TX commits, in caller — match create-plant.ts:295-315 outer-tx pattern)
        const postCommit = async () => {
          try {
            const ph = getPostHog();
            if (ph) {
              ph.capture({
                distinctId: userId,
                event: "identification_started",
                properties: {
                  provider: routerResult.ok ? routerResult.result.provider : routerResult.provider,
                  photo_count: photos.length,
                },
              });
              ph.capture({
                distinctId: userId,
                event: "identification_completed",
                properties: routerResult.ok
                  ? {
                      provider: routerResult.result.provider,
                      model: routerResult.result.model,
                      status: "success",
                      result_count: routerResult.result.results.length,
                      latency_ms: routerResult.result.latencyMs,
                      above_threshold: routerResult.result.results.length,
                    }
                  : {
                      provider: routerResult.provider,
                      model: "",
                      status: "failed",
                      failure_reason: routerResult.failureReason,
                      result_count: 0,
                      latency_ms: 0,
                      above_threshold: 0,
                    },
              });
              await ph.shutdown();
            }
          } catch (err) {
            Sentry.captureException(err, { tags: { surface: "identify-posthog" } });
          }

          if (!routerResult.ok) {
            try {
              await inngest.send({
                name: "identification.failed",
                data: {
                  identificationId: insertedRow.id, userId,
                  provider: routerResult.provider,
                  failureReason: routerResult.failureReason,
                  latencyMs: 0,
                  failedAt: new Date().toISOString(),
                } satisfies IdentificationFailedPayload,
              });
            } catch (err) {
              Sentry.captureException(err, { tags: { surface: "identify-inngest-failed" } });
            }
          }
          // identification.succeeded fires NOT here but in 06-09 confirm-identification (D-18)
        };

        const result: IdentifyResult = routerResult.ok
          ? { ok: true,
              body: {
                identification: toIdentificationSnakeCase(insertedRow),
                results: routerResult.result.results.map(toResultSnakeCase),
              },
              postCommit,
            }
          : { ok: false,
              code: mapRouterFailureToErrorCode(routerResult.failureReason),
              reason: humanReason(routerResult.failureReason),
              postCommit,
            };
        return result;
      });
    }

    function mapRouterFailureToErrorCode(reason: IdentificationFailureReason): ErrorCode {
      switch (reason) {
        case "timeout": return ErrorCode.Timeout;
        case "provider_unavailable":
        case "cost_ceiling_reached":
        case "breaker_open":
        case "invalid_response":
          return ErrorCode.ProviderUnavailable;
      }
    }

    function humanReason(reason: IdentificationFailureReason): string {
      // Internal — never returned to client (clients see error.message via errorResponse). Useful for Sentry tags.
      return reason;
    }

    function toResultSnakeCase(r) {
      return {
        species_name: r.speciesName,
        scientific_name: r.scientificName,
        confidence: r.confidence,
        provider_species_id: r.providerSpeciesId,
      };
    }
    ```

## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| Route → use-case | Photos arrive as `Buffer[]` AND `storedUrl[]` (already uploaded by route); consent gate, cap check, advisory lock all enforced inside the use-case |
| Use-case → DB (TX) | All cap math + Identification row insert happen inside `withUnitOfWork`; RLS engages |
| Use-case → Router → Provider HTTPS | Photos cross the trust boundary to Plant.id / OpenAI; LGPD Art. 33 consent gate is the legal authorization |
| Use-case → Inngest event bus (post-commit) | Post-commit emission of `identification.failed`; no PII payload |
