Author `deploy-preview.yml`. On every PR synchronize: create a Supabase branch DB, apply migrations, build via Vercel CLI, deploy preview, run Playwright smoke against the preview URL, comment the URL back on the PR. Source maps upload natively via `withSentryConfig` when `pnpm build` runs (D-19 OVERRIDE — no separate sentry-cli step).

Purpose: SC-2c verification — Playwright runs against a real preview URL bound to a per-PR Supabase branch. `deploy-preview-cleanup.yml` (plan 16) removes these on PR close (SC-2d).

**R7 mid-run cleanup reasoning:** If `deploy-preview.yml` dies mid-run (after branch creation, before Vercel deploy), the per-PR branch DB would be orphaned until the PR closes. Plan 16's `deploy-preview-cleanup.yml` is intentionally idempotent (`continue-on-error: true` on both delete steps, and the delete commands tolerate non-existent targets — see R8 for the visibility fix). This means a mid-run failure is self-healing once the PR closes: the cleanup workflow runs, finds a branch to delete, and succeeds. No explicit `if: always()` cleanup step is needed in this workflow as long as plan 16's cleanup remains idempotent. Executor: verify plan 16's idempotency assertions (R8) are intact before considering this task complete.

  ```yaml
  name: Deploy Preview

  on:
    pull_request:
      types: [opened, synchronize, reopened]

  concurrency:
    # R7 — PR-scoped concurrency group so rapid pushes to the same PR cancel any
    # in-flight preview deploy (otherwise the second run races with the first
    # against the same Supabase branch DB).
    group: preview-pr-${{ github.event.pull_request.number }}
    cancel-in-progress: true

  jobs:
    deploy:
      runs-on: ubuntu-latest
      timeout-minutes: 20
      environment: preview

      env:
        SUPABASE_ACCESS_TOKEN: ${{ secrets.SUPABASE_ACCESS_TOKEN }}
        SUPABASE_PROJECT_REF: ${{ secrets.SUPABASE_PROJECT_REF }}
        VERCEL_TOKEN: ${{ secrets.VERCEL_TOKEN }}
        VERCEL_ORG_ID: ${{ secrets.VERCEL_ORG_ID }}
        VERCEL_PROJECT_ID: ${{ secrets.VERCEL_PROJECT_ID }}
        SENTRY_AUTH_TOKEN: ${{ secrets.SENTRY_AUTH_TOKEN }}
        SENTRY_ORG: ${{ secrets.SENTRY_ORG }}
        SENTRY_PROJECT: ${{ secrets.SENTRY_PROJECT }}
        NEXT_PUBLIC_SENTRY_DSN: ${{ secrets.NEXT_PUBLIC_SENTRY_DSN }}
        NEXT_PUBLIC_POSTHOG_KEY: ${{ secrets.NEXT_PUBLIC_POSTHOG_KEY }}
        NEXT_PUBLIC_POSTHOG_HOST: https://us.i.posthog.com
        IDENTIFICATION_PROVIDER_MODE: stub   # Preview is ALWAYS stub (INFRA-16)

      steps:
        - name: Checkout
          uses: actions/checkout@v5
          with:
            fetch-depth: 0

        - name: Setup pnpm
          uses: pnpm/action-setup@v4
          with:
            version: 9

        - name: Setup Node
          uses: actions/setup-node@v4
          with:
            node-version-file: .nvmrc
            cache: pnpm

        - name: Setup Supabase CLI
          # R7 — pin to a verified version rather than `latest` so a future Supabase CLI
          # change cannot silently break the branching flow. Bump intentionally during revisions.
          uses: supabase/setup-cli@v1
          with:
            version: 1.219.2

        - name: Install deps
          run: pnpm install --frozen-lockfile

        - name: Create Supabase branch for PR
          id: branch
          # R7 — explicit exit-code handling instead of blanket `|| true`. We capture stderr
          # and distinguish "branch already exists" (continue) from genuine failures (abort).
          run: |
            BRANCH_NAME="pr-${{ github.event.number }}"
            set +e
            CREATE_OUT=$(supabase branches create "$BRANCH_NAME" --experimental --project-ref "$SUPABASE_PROJECT_REF" 2>&1)
            CREATE_RC=$?
            set -e
            if [ $CREATE_RC -ne 0 ]; then
              if echo "$CREATE_OUT" | grep -q "already exists"; then
                echo "Branch $BRANCH_NAME already exists — continuing to wait for ACTIVE_HEALTHY."
              else
                echo "::error::supabase branches create failed with rc=$CREATE_RC"
                echo "$CREATE_OUT"
                exit $CREATE_RC
              fi
            fi
            # Wait for branch to become ACTIVE_HEALTHY (up to 120s)
            for i in {1..24}; do
              STATUS=$(supabase branches get "$BRANCH_NAME" --experimental --project-ref "$SUPABASE_PROJECT_REF" -o json | jq -r '.status // empty')
              [ "$STATUS" = "ACTIVE_HEALTHY" ] && break
              sleep 5
            done
            # Capture branch DB URL
            DB_URL=$(supabase branches get "$BRANCH_NAME" --experimental --project-ref "$SUPABASE_PROJECT_REF" -o json | jq -r '.db_url')
            echo "::add-mask::$DB_URL"
            echo "db_url=$DB_URL" >> "$GITHUB_OUTPUT"

        - name: Apply migrations (drizzle-kit)
          env:
            DATABASE_URL: ${{ steps.branch.outputs.db_url }}
          run: pnpm exec drizzle-kit migrate

        - name: Vercel pull (preview env)
          run: pnpm exec vercel pull --yes --environment=preview --token=$VERCEL_TOKEN

        - name: Vercel build (withSentryConfig uploads source maps post-build — D-19 OVERRIDE)
          run: pnpm exec vercel build --token=$VERCEL_TOKEN
          env:
            VERCEL_GIT_COMMIT_SHA: ${{ github.sha }}

        - name: Vercel deploy --prebuilt
          id: deploy
          run: |
            URL=$(pnpm exec vercel deploy --prebuilt --token=$VERCEL_TOKEN)
            echo "url=$URL" >> "$GITHUB_OUTPUT"

        - name: Install Playwright browsers (Pitfall 7)
          run: pnpm exec playwright install --with-deps chromium

        - name: Run Playwright smoke against preview
          env:
            PLAYWRIGHT_TEST_BASE_URL: ${{ steps.deploy.outputs.url }}
          run: pnpm test:e2e

        - name: Upload Playwright report on failure
          if: failure()
          uses: actions/upload-artifact@v4
          with:
            name: playwright-report-${{ github.event.number }}
            path: playwright-report/
            retention-days: 7

        - name: Comment preview URL on PR
          uses: marocchino/sticky-pull-request-comment@v2
          with:
            header: preview
            message: |
              **Preview:** ${{ steps.deploy.outputs.url }}
              **Supabase branch:** `pr-${{ github.event.number }}`
  ```
