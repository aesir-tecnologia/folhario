export interface ListByUserCursor { createdAt: string; id: string; }
export interface ListByUserOptions { cursor?: ListByUserCursor; limit?: number; plantId?: string; }

export async function listByUser(db, userId, options): Promise<IdentificationRow[]> {
  const limit = Math.min(Math.max(options.limit ?? 20, 1), 200);
  const baseFilter = options.plantId
    ? and(eq(identifications.userId, userId), eq(identifications.plantId, options.plantId))
    : eq(identifications.userId, userId);
  const cursor = options.cursor;
  const where = cursor
    ? and(baseFilter, or(
        lt(identifications.createdAt, cursor.createdAt),
        and(eq(identifications.createdAt, cursor.createdAt), lt(identifications.id, cursor.id)),
      ))
    : baseFilter;
  return db.select().from(identifications).where(where)
    .orderBy(desc(identifications.createdAt), desc(identifications.id))
    .limit(limit);
}

const cache = new Map<"trial" | "paid", Cached>();
const TTL_MS = 30_000;

export async function findByTier(db, tier): Promise<IdentificationLimitRow> {
  const now = Date.now();
  const hit = cache.get(tier);
  if (hit && hit.expiresAt > now) return hit.row;
  const [row] = await db.select().from(identificationLimits).where(eq(identificationLimits.tier, tier)).limit(1);
  if (!row) throw new Error(`identification_limits row missing for tier=${tier}`);
  cache.set(tier, { row, expiresAt: now + TTL_MS });
  return row;
}

export function __resetCacheForTests() { cache.clear(); }

export async function hasActiveConsent(
  db: ConsentLogsDb,
  userId: string,
  purpose: ConsentPurpose,
): Promise<boolean> {
  const [latest] = await db
    .select({ revokedAt: consentLogs.revokedAt })
    .from(consentLogs)
    .where(and(eq(consentLogs.userId, userId), eq(consentLogs.purpose, purpose)))
    .orderBy(desc(consentLogs.grantedAt))
    .limit(1);
  return latest != null && latest.revokedAt == null;
}
