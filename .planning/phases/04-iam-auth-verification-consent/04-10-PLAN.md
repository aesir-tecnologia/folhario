    **Codex HIGH #4 + D-31 fix — `src/contexts/iam/api/components/use-auth-form.ts`** (CLIENT hook):
    ```typescript
    "use client";

    import { useState } from "react";

    /**
     * Codex HIGH #4 + D-31 fix: shared client-side hook for auth form submission.
     * - Submits via fetch() with Content-Type: application/json
     * - Maps server error codes (e.g., "invalid_credentials", "rate_limited") to next-intl message keys
     * - Returns { isSubmitting, error, submit, errorCode }
     *
     * Each form component (signup, login, etc.) calls submit({endpoint, body}) on form onSubmit.
     */
    export type AuthFormError =
      | { code: "validation_failed"; field?: string; message: string }
      | { code: "invalid_credentials"; message: string }
      | { code: "invalid_partner_code"; message: string }
      | { code: "rate_limited"; message: string }
      | { code: "internal_error"; message: string };

    export function useAuthForm() {
      const [isSubmitting, setIsSubmitting] = useState(false);
      const [error, setError] = useState<AuthFormError | null>(null);

      async function submit(opts: {
        endpoint: string;
        method?: "POST" | "PATCH";
        body: Record<string, unknown>;
        onSuccess?: (data: any) => void;
      }) {
        setIsSubmitting(true);
        setError(null);
        try {
          const resp = await fetch(opts.endpoint, {
            method: opts.method ?? "POST",
            headers: { "Content-Type": "application/json" },
            body: JSON.stringify(opts.body),
          });
          if (!resp.ok) {
            const errorBody = await resp.json().catch(() => ({ error: { code: "internal_error", message: "Erro desconhecido." } }));
            setError(errorBody.error as AuthFormError);
            return;
          }
          const data = await resp.json();
          opts.onSuccess?.(data);
        } catch (e: any) {
          setError({ code: "internal_error", message: e?.message ?? "Erro de rede." });
        } finally {
          setIsSubmitting(false);
        }
      }

      return { isSubmitting, error, errorCode: error?.code, submit, clearError: () => setError(null) };
    }
    ```

    ```tsx
    "use client";
    import { useState } from "react";
    import { useTranslations } from "next-intl";

    export function ResendVerificationButton() {
      const t = useTranslations("auth.unverifiedBlocker");
      const [cooldown, setCooldown] = useState(0);
      const [isSending, setIsSending] = useState(false);

      async function handleClick() {
        if (cooldown > 0 || isSending) return;
        setIsSending(true);
        try {
          const resp = await fetch("/api/v1/iam/resend-verification", {
            method: "POST", headers: { "Content-Type": "application/json" }, body: "{}",
          });
          if (resp.status === 429) {
            // Rate-limited — start 60s cooldown.
            setCooldown(60);
            const interval = setInterval(() => {
              setCooldown(c => {
                if (c <= 1) { clearInterval(interval); return 0; }
                return c - 1;
              });
            }, 1000);
          }
        } finally { setIsSending(false); }
      }

      return (
        <button type="button" onClick={handleClick} disabled={cooldown > 0 || isSending}
          style={{
            width: "100%", maxWidth: 320, height: 48,
            backgroundColor: cooldown > 0 ? "#5A6358" : "#1F4D35",
            color: "#FFFDF7", fontFamily: "'Plus Jakarta Sans', system-ui, sans-serif",
            fontSize: 16, fontWeight: 600, border: "none", borderRadius: 8,
            cursor: cooldown > 0 ? "not-allowed" : "pointer",
          }}>
          {isSending ? t("resending") : cooldown > 0 ? t("cooldown", { seconds: cooldown }) : t("resend")}
        </button>
      );
    }
    ```
