    Implement the function in `src/contexts/identification/inngest/functions.ts`:

    ```typescript
    import { inngest } from "@shared/inngest/client";
    import { resendAdapter } from "@contexts/notifications/infrastructure/resend-adapter";
    import { serverEnv } from "@shared/config/server-env";
    import type { ProviderCeilingReachedPayload } from "@contexts/identification/domain/events";

    const notifyCeiling = inngest.createFunction(
      { id: "identification-notify-ceiling", retries: 3 },
      { event: "provider.ceiling_reached" },
      async ({ event, step }) => {
        const data = event.data as ProviderCeilingReachedPayload;
        if (!serverEnv.OPERATOR_ALERT_EMAIL) {
          // Dev-mode fallback (no OPERATOR_ALERT_EMAIL configured); matches resend-adapter dev fallback shape
          console.warn("[notifyCeiling] OPERATOR_ALERT_EMAIL not set; skipping operator alert", {
            provider: data.provider, purpose: data.purpose, utcDate: data.utcDate,
          });
          return { messageId: "dev-skipped" };
        }
        const pct = ((data.estimatedCostCents / data.capCents) * 100).toFixed(1);
        const result = await step.run("send-operator-email", async () =>
          resendAdapter.send({
            from: serverEnv.RESEND_FROM_ADDRESS, // verify exact env var name in server-env.ts; if it's RESEND_FROM use that
            to: serverEnv.OPERATOR_ALERT_EMAIL!,
            subject: `[Folhário] ${data.provider} ${data.purpose} at ${pct}% daily ceiling — ${data.utcDate}`,
            html: `<p>Provider <strong>${data.provider}</strong> (${data.purpose}) reached <strong>${pct}%</strong> of daily ceiling on ${data.utcDate}.</p>` +
                  `<p>Spend: ${data.estimatedCostCents} / ${data.capCents} cents.</p>` +
                  `<p>This alert is debounced once per UTC day per (provider, purpose) via provider_budgets.last_alerted_at.</p>`,
            templateName: "operator-ceiling-alert",
          })
        );
        return { messageId: result.id };
      }
    );

    export const identificationFunctions = [notifyCeiling];
    ```

    Note: `resendAdapter.send` is the entry point — DO NOT instantiate `new Resend(serverEnv.RESEND_API_KEY)` directly. The adapter handles env-gate + dev-fallback (resend-adapter.ts:21-37 pattern).
