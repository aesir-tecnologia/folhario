    ```typescript
    // Phase 4 AUTH-13 + Codex HIGH #3 (AuthAdapter boundary).
    import { authAdapter } from "@contexts/iam/infrastructure/auth-adapter";

    export type ChangePasswordResult =
      | { kind: "success" }
      | { kind: "invalid_credentials" }
      | { kind: "oauth_only" };

    export async function changePassword(opts: {
      userId: string;
      email: string;
      currentPassword: string;
      newPassword: string;
      hasPassword: boolean;
    }): Promise<ChangePasswordResult> {
      // AUTH-13 + UI-SPEC §4: OAuth-only users cannot change password.
      if (!opts.hasPassword) return { kind: "oauth_only" };

      // Reverify current password via the AuthAdapter (Codex HIGH #3).
      const verify = await authAdapter.signInWithPassword({
        email: opts.email,
        password: opts.currentPassword,
      });
      if (!verify.ok) return { kind: "invalid_credentials" };

      // Set the new password via the adapter (does not invalidate existing JWTs — same as AUTH-12).
      const update = await authAdapter.adminUpdatePassword({ userId: opts.userId, newPassword: opts.newPassword });
      if (!update.ok) throw new Error(`adminUpdatePassword failed: ${update.reason}`);

      return { kind: "success" };
    }
    ```

    ```typescript
    // Phase 4 AUTH-03 + D-04 + D-32 + Codex HIGH #2 (db.transaction).
    import { db } from "@contexts/iam/infrastructure/db/client";
    import { getUserById, setOauthCompletionFields } from "@contexts/iam/infrastructure/db/users";
    import { isPartnerCodeActive } from "@contexts/iam/infrastructure/db/partner-store";
    import { getCurrentPolicyVersion } from "@contexts/iam/infrastructure/db/policy-versions";
    import { insertSignupConsents } from "@contexts/iam/infrastructure/db/consent-logs";
    import { insertTrialingSubscription } from "@contexts/iam/infrastructure/db/subscriptions";
    import { captureSignupCompleted } from "@contexts/iam/infrastructure/posthog-bridge";
    import type { OAuthComplete } from "@contexts/iam/domain/schemas";

    export type OauthCompleteResult =
      | { kind: "success" }
      | { kind: "already_completed" }
      | { kind: "invalid_partner_code" };

    export async function completeOauthSignup(
      input: OAuthComplete & { userId: string; email: string },
    ): Promise<OauthCompleteResult> {
      const existing = await getUserById(input.userId);
      if (!existing) throw new Error("oauth-complete: user not found in public.users");
      if (existing.ageConfirmedAt) return { kind: "already_completed" };

      // D-32: validate partner_code against partner_store (same path as signup).
      const partnerCodeRaw = (input.partner_code ?? "").trim();
      let trialSource: "organic" | "partner" = "organic";
      if (partnerCodeRaw.length > 0) {
        const isActive = await isPartnerCodeActive(partnerCodeRaw);
        if (!isActive) return { kind: "invalid_partner_code" };
        trialSource = "partner";
      }

      // Codex HIGH #2: ALL writes inside ONE db.transaction(...).
      await db.transaction(async (tx) => {
        const policy = await getCurrentPolicyVersion(tx);
        if (!policy) throw new Error("oauth-complete: no current policy_versions row");

        await setOauthCompletionFields(input.userId, {
          timezone: input.timezone,
          partnerCode: partnerCodeRaw.length > 0 ? partnerCodeRaw : null,
        }, tx);
        await insertSignupConsents({ userId: input.userId, policyVersionId: policy.id }, tx);
        await insertTrialingSubscription({
          userId: input.userId,
          partnerCode: partnerCodeRaw.length > 0 ? partnerCodeRaw : null,
          trialSource,
        }, tx);
      });

      // Pitfall 8: signup_completed fires HERE (OAuth users skip the verify route).
      await captureSignupCompleted(input.userId);

      return { kind: "success" };
    }
    ```
