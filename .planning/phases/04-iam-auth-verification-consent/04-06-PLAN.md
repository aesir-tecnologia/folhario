    ```typescript
    // Phase 4 AUTH-01/04/06/07/08. Replaces Plan 03's stub.
    import { eq, sql } from "drizzle-orm";
    import { db } from "@contexts/iam/infrastructure/db/client";
    import { users } from "@contexts/iam/infrastructure/db/schema";
    import type { User } from "@contexts/iam/domain/schemas";

    /**
     * Reads public.users joined with auth.users to derive hasPassword.
     * D-22: email_verified_at comes from public.users — never auth.users.email_confirmed_at.
     */
    export async function getUserById(id: string): Promise<User | null> {
      const rows = await db.execute<{ id: string; email: string; email_verified_at: Date | null; age_confirmed_at: Date | null; timezone: string; partner_code: string | null; has_password: boolean }>(
        sql`SELECT pu.id, pu.email, pu.email_verified_at, pu.age_confirmed_at, pu.timezone, pu.partner_code,
                   (au.encrypted_password IS NOT NULL) AS has_password
            FROM public.users pu
            JOIN auth.users au ON au.id = pu.id
            WHERE pu.id = ${id}`
      );
      const row = rows[0];
      if (!row) return null;
      return {
        id: row.id,
        email: row.email,
        emailVerifiedAt: row.email_verified_at,
        ageConfirmedAt: row.age_confirmed_at,
        timezone: row.timezone,
        partnerCode: row.partner_code,
        hasPassword: row.has_password,
      };
    }

    export async function getUserByEmail(email: string): Promise<User | null> {
      const rows = await db.execute<{ id: string; email: string; email_verified_at: Date | null; age_confirmed_at: Date | null; timezone: string; partner_code: string | null; has_password: boolean }>(
        sql`SELECT pu.id, pu.email, pu.email_verified_at, pu.age_confirmed_at, pu.timezone, pu.partner_code,
                   (au.encrypted_password IS NOT NULL) AS has_password
            FROM public.users pu
            JOIN auth.users au ON au.id = pu.id
            WHERE pu.email = ${email}`
      );
      const row = rows[0];
      if (!row) return null;
      return {
        id: row.id,
        email: row.email,
        emailVerifiedAt: row.email_verified_at,
        ageConfirmedAt: row.age_confirmed_at,
        timezone: row.timezone,
        partnerCode: row.partner_code,
        hasPassword: row.has_password,
      };
    }

    export async function insertPublicUser(input: {
      id: string; email: string; age_confirmed_at: Date; timezone: string; partner_code: string | null;
    }): Promise<void> {
      // Idempotent: Phase 2 D-35 trigger may have already created a minimal row.
      await db.execute(
        sql`INSERT INTO public.users (id, email, age_confirmed_at, timezone, partner_code)
            VALUES (${input.id}, ${input.email}, ${input.age_confirmed_at}, ${input.timezone}, ${input.partner_code})
            ON CONFLICT (id) DO UPDATE SET
              email = EXCLUDED.email,
              age_confirmed_at = EXCLUDED.age_confirmed_at,
              timezone = EXCLUDED.timezone,
              partner_code = EXCLUDED.partner_code`
      );
    }

    export async function setEmailVerifiedAt(id: string, when: Date): Promise<void> {
      await db.update(users).set({ email_verified_at: when }).where(eq(users.id, id));
    }
    ```

    **`verification-tokens.ts`**:
    ```typescript
    // Phase 4 D-06 + AUTH-01/04. Token hash storage; raw exists only in URL.
    import { eq, and, isNull, sql } from "drizzle-orm";
    import { db } from "@contexts/iam/infrastructure/db/client";
    import { emailVerificationTokens } from "@contexts/iam/infrastructure/db/schema";
    import { mintToken } from "@shared/crypto/tokens";
    import { createHash } from "node:crypto";

    export async function revokeUnusedVerificationTokens(userId: string): Promise<void> {
      await db.update(emailVerificationTokens)
        .set({ consumed_at: new Date() })
        .where(and(eq(emailVerificationTokens.user_id, userId), isNull(emailVerificationTokens.consumed_at)));
    }

    export async function mintVerificationToken(opts: {
      userId: string;
      sentToEmail: string;
    }): Promise<{ tokenId: string; rawToken: string }> {
      // D-06 single active token: revoke prior first.
      await revokeUnusedVerificationTokens(opts.userId);

      const { raw, hash } = mintToken();
      const expiresAt = new Date(Date.now() + 24 * 60 * 60 * 1000);  // 24h per D-06

      const [row] = await db.insert(emailVerificationTokens).values({
        user_id: opts.userId,
        token_hash: hash,
        expires_at: expiresAt,
        sent_to_email: opts.sentToEmail,
      }).returning({ id: emailVerificationTokens.id });

      return { tokenId: row!.id, rawToken: raw };
    }

    export async function consumeVerificationToken(rawToken: string): Promise<{ userId: string } | null> {
      const hash = createHash("sha256").update(rawToken).digest("hex");
      // FOR UPDATE locks the row; UPDATE SET consumed_at in same tx.
      const result = await db.execute<{ user_id: string }>(
    ```

    ```typescript
    import { errorResponse, ErrorCode } from "@shared/config/errors";
    import { requireApiUser } from "@shared/api/auth";
    import { resendVerification } from "@contexts/iam/application/resend-verification";

      if ("error" in result) return result.error;
      const { user } = result;

      // Idempotent if already verified.
      if (user.emailVerifiedAt) {
        return NextResponse.json({ ok: true, message: "E-mail já verificado." });
      }

      // D-08: per-user 1/min throttle. Use user.id as the "ip" key in auth_throttle.
      const { count } = await bumpThrottle(`user:${user.id}`, "resend-verification");

        return errorResponse(ErrorCode.InternalError, "Não foi possível reenviar agora.");

    }
    ```
