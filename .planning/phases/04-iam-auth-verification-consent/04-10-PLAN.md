    **Modify `src/app/layout.tsx`** — implement resolved Q4 gate ordering. Preserve `NextIntlClientProvider` and `PostHogProvider` wraps:
    ```tsx
    // Phase 4 D-21 + Resolved Q4. Root layout server component implements the gate.
    import { getLocale, getMessages } from "next-intl/server";
    import { NextIntlClientProvider } from "next-intl";
    import { redirect } from "next/navigation";
    import { headers } from "next/headers";
    import type { ReactNode } from "react";
    import { PostHogProvider } from "@/app/posthog-provider";
    import { getCurrentUserOrNull } from "@contexts/iam/application/current-user";
    import { isUnverifiedAllowed } from "@shared/api/auth";
    import { UnverifiedBlocker } from "@contexts/iam/api/components/unverified-blocker";

    export default async function RootLayout({ children }: { children: ReactNode }) {
      const locale = await getLocale();
      const messages = await getMessages();

      // Resolved Q4 gate ordering:
      // (a) JWT check
      // (b) age_confirmed_at IS NULL → /auth/oauth-complete
      // (c) email_verified_at IS NULL → UnverifiedBlocker
      // (d) authorized children
      const user = await getCurrentUserOrNull();
      const headersList = await headers();
      const pathname = headersList.get("x-pathname") ?? "/";  // proxy may set this; fallback safe

      // Read-only paths (auth flows, static, infra) skip the gate entirely.
      const isAllowed = isUnverifiedAllowed(pathname);

      if (user && !isAllowed) {
        // Resolved Q4 (b): if OAuth is incomplete, redirect.
        if (!user.ageConfirmedAt && pathname !== "/auth/oauth-complete") {
          redirect("/auth/oauth-complete");
        }
        // Resolved Q4 (c): if email not verified, render the blocker.
        if (!user.emailVerifiedAt) {
          return (
            <html lang={locale}>
              <head>
                <link rel="manifest" href="/manifest.webmanifest" />
              </head>
              <body>
                <NextIntlClientProvider locale={locale} messages={messages}>
                  <PostHogProvider>
                    <UnverifiedBlocker email={user.email} />
                  </PostHogProvider>
                </NextIntlClientProvider>
              </body>
            </html>
          );
        }
      }

      // Resolved Q4 (d): authorized children.
      return (
        <html lang={locale}>
          <head>
            <link rel="manifest" href="/manifest.webmanifest" />
          </head>
          <body>
            <NextIntlClientProvider locale={locale} messages={messages}>
              <PostHogProvider>{children}</PostHogProvider>
            </NextIntlClientProvider>
          </body>
        </html>
      );
    }
    ```

    NOTE on x-pathname: Next 16 doesn't expose pathname in Server Components by default. Workaround: the proxy (or middleware) can set `x-pathname` header on every request. If the proxy doesn't set it, the gate falls through to "/" as default and the allowlist matches "/" (which is NOT allowed for unverified users — but Phase 4 root "/" landing always renders the blocker for unverified users, which is the desired behavior).
