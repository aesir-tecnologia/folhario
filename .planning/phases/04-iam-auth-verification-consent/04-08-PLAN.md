<objective>
Implement the password reset flow per CONTEXT D-09, D-10, D-11. The request endpoint is a **thin `inngest.send` wrapper** — sub-millisecond, constant-time response (no DB lookup, no token mint, no conditional logic) per D-11 anti-enumeration timing-attack defense. The actual user lookup + token mint + email dispatch happens **asynchronously inside the `iam/password-reset-requested` Inngest function** (REPLACES the Plan 04 stub — same pattern as Plan 05 replacing the notifications/send-email stub). The consume endpoint validates the SHA-256 token in a transaction and updates the password via `authAdapter.adminUpdatePassword` (Codex HIGH #3) — without invalidating existing JWTs (per AUTH-12).

Purpose: AUTH-11 (always-200 anti-enumeration) and AUTH-12 (token consume + admin.updateUserById; existing JWTs valid). Resolved by D-11's enqueue-first pattern: the public endpoint is fast, unconditional, and constant-time; user existence is checked async inside the Inngest function. This eliminates the response-latency timing side-channel (existing-user lookup ~50–200ms vs non-existing ~10–20ms) that an inline-lookup pattern would expose.

**D-11 vs D-16 Decision resolution (per user directive 2026-04-26):** A prior planner pass interpreted D-16 as a hard 8-function cap and inlined the lookup into the route handler — which re-introduced the timing leak D-11 was designed to defend against. The user resolved the conflict by reframing INFRA-10 as **"8 PRD §3 MVP + 1 Phase-4 anti-enumeration add"** — restoring `iam/password-reset-requested` as the 9th registered function. This plan ships its real impl (replacing the Plan 04 stub).

**Codex HIGH #2 fix:** consumePasswordReset wraps consumeResetToken + authAdapter.adminUpdatePassword in `db.transaction(...)`. If the password update fails, the token consume is rolled back. The same pattern applies inside the iam/password-reset-requested Inngest function: revoke prior unused tokens + mint fresh token are wrapped in a single db.transaction (D-25 atomicity).

**Codex HIGH #3 fix:** consumePasswordReset and the iam/password-reset-requested function both use `authAdapter` (NOT `supabase.auth.*` directly).

**Codex HIGH #6 fix:** Playwright E2E covers cookie-bearing reset → login flow.

**Codex HIGH #8 fix:** The OUTER `iam/password-reset-requested` event id is `password-reset-request/{email}/{minute-bucket}` (1-minute window). This dedupes accidental rapid-fire from double-clicks while still allowing a legitimate new request after the 1h token expiry. The INNER `notifications/email.requested` event id stays `password-reset/{tokenId}` for per-token dedup (token id is unique per mint).

**D-31 fix:** Both routes are JSON-only — `request.json()`, never `request.formData()`.

Output: 1 new repository (reset-tokens), 2 application use-cases, 1 Inngest function impl (replaces stub), 2 route handlers, 2 vitest integration tests (full flow + constant-time route-handler timing), 1 Playwright E2E (cookie + login-after-reset).

    // Phase 4 D-09 + AUTH-11. Token hash storage; raw exists only in URL.
    import { eq, and, isNull, sql } from "drizzle-orm";
    import { db } from "@contexts/iam/infrastructure/db/client";
    import { passwordResetTokens } from "@contexts/iam/infrastructure/db/schema";
    import { mintToken } from "@shared/crypto/tokens";
    import { createHash } from "node:crypto";
    import type { DbOrTx } from "@contexts/iam/infrastructure/db/types";

    export async function revokeUnusedResetTokens(userId: string, dbOrTx: DbOrTx = db): Promise<void> {
      await dbOrTx.update(passwordResetTokens)
        .set({ consumed_at: new Date() })
        .where(and(eq(passwordResetTokens.user_id, userId), isNull(passwordResetTokens.consumed_at)));
    }

    export async function mintResetToken(opts: {
      userId: string;
      sentToEmail: string;
    }, dbOrTx: DbOrTx = db): Promise<{ tokenId: string; rawToken: string }> {
      await revokeUnusedResetTokens(opts.userId, dbOrTx);
      const { raw, hash } = mintToken();
      const expiresAt = new Date(Date.now() + 60 * 60 * 1000);  // D-09: 1h per AUTH-11
      const [row] = await dbOrTx.insert(passwordResetTokens).values({
        user_id: opts.userId,
        token_hash: hash,
        expires_at: expiresAt,
        sent_to_email: opts.sentToEmail,
      }).returning({ id: passwordResetTokens.id });
      return { tokenId: row!.id, rawToken: raw };
    }

    export async function consumeResetToken(rawToken: string, dbOrTx: DbOrTx = db): Promise<{ userId: string } | null> {
      const hash = createHash("sha256").update(rawToken).digest("hex");
      const result = await dbOrTx.execute<{ user_id: string }>(
        sql`WITH t AS (
              SELECT id, user_id FROM public.password_reset_tokens
              WHERE token_hash = ${hash} AND consumed_at IS NULL AND expires_at > now()
              FOR UPDATE
            )
            UPDATE public.password_reset_tokens prt SET consumed_at = now()
            FROM t
            WHERE prt.id = t.id
            RETURNING prt.user_id`
      );
      const row = result[0];
      return row ? { userId: row.user_id } : null;
    }

    // INVOKED FROM INSIDE the iam/password-reset-requested Inngest function (NOT from the route handler).
    // The route handler is a thin inngest.send wrapper — sub-millisecond, constant-time response.
    // All conditional logic (lookup, branch, mint, dispatch) happens async here for D-11's
    // anti-enumeration timing-attack defense.
    import { getTranslations } from "next-intl/server";
    import { inngest } from "@shared/inngest/client";
    import { db } from "@contexts/iam/infrastructure/db/client";
    import { getUserByEmail } from "@contexts/iam/infrastructure/db/users";
    import { mintResetToken } from "@contexts/iam/infrastructure/db/reset-tokens";

    export async function requestPasswordReset(args: {
      email: string;
      requestUrl: string;
    }): Promise<void> {
      const user = await getUserByEmail(args.email);
      if (!user) return;                       // anti-enumeration: no email sent
      if (!user.hasPassword) return;           // OAuth-only: no email sent (anti-enumeration)

      // D-25 atomicity: revoke prior + mint fresh in one tx so a partial failure can't
      // leave orphaned revoked tokens with no replacement.
      const { tokenId, rawToken } = await db.transaction(async (tx) => {
        return await mintResetToken({ userId: user.id, sentToEmail: args.email }, tx);
      });

      const resetUrl = new URL(`/auth/reset?token=${rawToken}`, args.requestUrl).toString();
      const t = await getTranslations("email.passwordReset");

      // Codex HIGH #8: inner event id is `password-reset/{tokenId}` — token-unique.
      // Outer iam/password-reset-requested event id (set in route handler) handles minute-bucket
      // dedup to collapse double-clicks.
      await inngest.send({
        id: `password-reset/${tokenId}`,
        name: "notifications/email.requested",
        data: {
          template: "password-reset",
          to: args.email,
          subject: t("subject"),
          props: { url: resetUrl, userEmail: args.email },
        },
      });
    }
