    ```typescript
    // Phase 4 AUTH-13. Reverify current password via signInWithPassword, then admin.updateUserById.
    // OAuth-only users blocked at endpoint with forbidden 403 (UI hides the form per UI-SPEC §4).
    import { getSupabaseServerClient } from "@contexts/iam/infrastructure/supabase-server";
    import { supabaseAdmin } from "@contexts/iam/infrastructure/supabase-admin";

    export type ChangePasswordResult =
      | { kind: "success" }
      | { kind: "invalid_credentials" }
      | { kind: "oauth_only" };

    export async function changePassword(opts: {
      userId: string;
      email: string;
      currentPassword: string;
      newPassword: string;
      hasPassword: boolean;
    }): Promise<ChangePasswordResult> {
      // AUTH-13 + UI-SPEC §4: OAuth-only users cannot change password.
      if (!opts.hasPassword) return { kind: "oauth_only" };

      // Reverify current password (Supabase signInWithPassword as a verification probe).
      const supabase = await getSupabaseServerClient();
      const { error: verifyError } = await supabase.auth.signInWithPassword({
        email: opts.email,
        password: opts.currentPassword,
      });
      if (verifyError) return { kind: "invalid_credentials" };

      // Set the new password via admin (does not invalidate existing JWTs — same as AUTH-12).
      const { error: updateError } = await supabaseAdmin.auth.admin.updateUserById(opts.userId, {
        password: opts.newPassword,
      });
      if (updateError) throw new Error(`updateUserById failed: ${updateError.message}`);

      return { kind: "success" };
    }
    ```

    ```typescript
    import { sql } from "drizzle-orm";
    import { db } from "@contexts/iam/infrastructure/db/client";
    import { getUserById } from "@contexts/iam/infrastructure/db/users";
    import { getCurrentPolicyVersion } from "@contexts/iam/infrastructure/db/policy-versions";
    import { insertSignupConsents } from "@contexts/iam/infrastructure/db/consent-logs";
    import { insertTrialingSubscription } from "@contexts/iam/infrastructure/db/subscriptions";
    import { captureSignupCompleted } from "@contexts/iam/infrastructure/posthog-bridge";
    import type { OAuthComplete } from "@contexts/iam/domain/schemas";

    export async function completeOauthSignup(
      input: OAuthComplete & { userId: string; email: string },
    ): Promise<{ kind: "success" } | { kind: "already_completed" }> {
      // Idempotent: if age_confirmed_at already set, OAuth-complete already ran.
      const existing = await getUserById(input.userId);
      if (!existing) throw new Error("oauth-complete: user not found in public.users");
      if (existing.ageConfirmedAt) return { kind: "already_completed" };

      const policy = await getCurrentPolicyVersion();
      if (!policy) throw new Error("oauth-complete: no current policy_versions row");

      // D-04: write age, timezone, partner_code, AND email_verified_at = now() in one statement.
      await db.execute(
        sql`UPDATE public.users SET
              age_confirmed_at = now(),
              timezone = ${input.timezone},
              partner_code = ${input.partner_code ?? null},
              email_verified_at = now()
            WHERE id = ${input.userId}`,
      );

      // D-24: same two ConsentLog rows as email+password signup.
      await insertSignupConsents({ userId: input.userId, policyVersionId: policy.id });

      // Initial trialing subscription (organic vs partner depending on partner_code).

      // Pitfall 8: signup_completed fires HERE (OAuth users skip the verify route).
      await captureSignupCompleted(input.userId);

      return { kind: "success" };
    }
    ```

    ```typescript
    import { NextResponse } from "next/server";
    import { z } from "zod";
    import { errorResponse, ErrorCode } from "@shared/config/errors";
    import { requireApiUser } from "@shared/api/auth";
    import { sql } from "drizzle-orm";
    import { db } from "@contexts/iam/infrastructure/db/client";
    import { getUserById } from "@contexts/iam/infrastructure/db/users";

    export async function GET(request: Request) {
      const r = await requireApiUser(request);
      if ("error" in r) return r.error;
      return NextResponse.json({ user: r.user });
    }

    const patchSchema = z.object({
      timezone: z.string().refine((v) => Intl.supportedValuesOf("timeZone").includes(v), { message: "invalid_timezone" }).optional(),
    }).strict();

    export async function PATCH(request: Request) {
      const r = await requireApiUser(request);
      if ("error" in r) return r.error;

      let body: unknown;
      try { body = await request.json(); } catch { return errorResponse(ErrorCode.ValidationFailed, "Corpo inválido."); }
      const parsed = patchSchema.safeParse(body);
      if (!parsed.success) return errorResponse(ErrorCode.ValidationFailed, "Validação falhou.");

      if (parsed.data.timezone) {
        await db.execute(sql`UPDATE public.users SET timezone = ${parsed.data.timezone} WHERE id = ${r.user.id}`);
      }
      const updated = await getUserById(r.user.id);
      return NextResponse.json({ user: updated });
    }
    ```

    ```typescript
    import { NextResponse } from "next/server";
    import { errorResponse, ErrorCode } from "@shared/config/errors";
    import { requireApiUser } from "@shared/api/auth";
    import { changePasswordSchema } from "@contexts/iam/domain/schemas";
    import { changePassword } from "@contexts/iam/application/change-password";

    export async function PATCH(request: Request) {
      const r = await requireApiUser(request);
      if ("error" in r) return r.error;

      let body: unknown;
      try { body = await request.json(); } catch { return errorResponse(ErrorCode.ValidationFailed, "Corpo inválido."); }
      const parsed = changePasswordSchema.safeParse(body);
      if (!parsed.success) return errorResponse(ErrorCode.ValidationFailed, "Validação falhou.");

      const result = await changePassword({
        userId: r.user.id,
        email: r.user.email,
        currentPassword: parsed.data.current_password,
        newPassword: parsed.data.new_password,
        hasPassword: r.user.hasPassword,
      });

      if (result.kind === "oauth_only") return errorResponse(ErrorCode.Forbidden, "Conta sem senha local.");
      if (result.kind === "invalid_credentials") return errorResponse(ErrorCode.InvalidCredentials, "Senha atual incorreta.");
      return NextResponse.json({ ok: true });
    }
    ```

    ```typescript
    // Phase 4 AUTH-03 + D-04 + Resolved Q4. OAuth callback completes code-for-session exchange.
    // Routing per resolved Q4: (a) JWT check → (b) age_confirmed_at IS NULL → /auth/oauth-complete → (c) /
    import { NextResponse } from "next/server";
    import { withThrottle } from "@shared/api/throttle";
    import { getSupabaseServerClient } from "@contexts/iam/infrastructure/supabase-server";
    import { getUserById } from "@contexts/iam/infrastructure/db/users";

    export async function GET(request: Request) {
      return withThrottle(request, "oauth-callback", "on-failure", async () => {
        const url = new URL(request.url);
        const code = url.searchParams.get("code");
        const oauthError = url.searchParams.get("error");

        if (oauthError || !code) {
          // 401 status triggers withThrottle "on-failure" increment.
          return NextResponse.redirect(new URL("/auth/login?error=oauth_failed", url), { status: 302 });
        }

        const supabase = await getSupabaseServerClient();
        const { error: exchangeError } = await supabase.auth.exchangeCodeForSession(code);
        if (exchangeError) {
          return NextResponse.redirect(new URL("/auth/login?error=oauth_failed", url), { status: 302 });
        }

        const { data: { user: supabaseUser } } = await supabase.auth.getUser();
        if (!supabaseUser) {
          return NextResponse.redirect(new URL("/auth/login?error=oauth_failed", url), { status: 302 });
        }
    ```
