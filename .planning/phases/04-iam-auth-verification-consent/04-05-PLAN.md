    ```typescript
    // Phase 4 NOTIF-01/02: render template name + props into a React element for Resend.
    import type { ReactElement } from "react";
    import type { TemplateName } from "@contexts/notifications/domain/events";
    import { VerificationEmail } from "@contexts/notifications/infrastructure/email-templates/verification";
    import { PasswordResetEmail } from "@contexts/notifications/infrastructure/email-templates/password-reset";
    import { WelcomeBackEmail } from "@contexts/notifications/infrastructure/email-templates/welcome-back";

    /**
     * Resolves a template name + props into a React element. Caller (Inngest function)
     * passes the result to resendAdapter.send({ react }).
     *
     * Type-narrowing: each template name is paired with its expected props shape via
     * the discriminated union in NotificationsEmailRequestedPayload. The `props` arg
     * here is intentionally `unknown` — runtime type-checking lives at the Inngest
     * boundary (Zod parse on event.data) before this is called.
     */
    export function renderEmail(
      template: TemplateName,
      props: unknown,
    ): { react: ReactElement } {
      switch (template) {
        case "verification":
          return { react: VerificationEmail(props as any) };
        case "password-reset":
          return { react: PasswordResetEmail(props as any) };
        case "welcome-back":
          return { react: WelcomeBackEmail(props as any) };
        default: {
          const exhaustive: never = template;
          throw new Error(`Unknown email template: ${exhaustive}`);
        }
      }
    }
    ```
